## 10. 因果机器学习：cross-fitting、overlap 与验证（Python）

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import GroupKFold
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier
from sklearn.metrics import roc_auc_score
from econml.dml import CausalForestDML

df = pd.read_stata(r"终版数据")
y = df["Y"].to_numpy(); t = df["T"].to_numpy()
X = df[["X1", "X2"]].to_numpy(); W = df[["W1", "W2"]].to_numpy()
groups = df["panel_id"].to_numpy()
cv = GroupKFold(n_splits=5)
prop = RandomForestClassifier(n_estimators=500, min_samples_leaf=20, random_state=20260927)
prop.fit(np.c_[X, W], t)
ps = prop.predict_proba(np.c_[X, W])[:, 1]
print(pd.Series(ps).describe(percentiles=[.01,.05,.95,.99]))

est = CausalForestDML(
    model_y=RandomForestRegressor(n_estimators=500, min_samples_leaf=10, random_state=20260927),
    model_t=RandomForestClassifier(n_estimators=500, min_samples_leaf=10, random_state=20260927),
    discrete_treatment=True, n_estimators=1000, min_samples_leaf=10,
    cv=cv, random_state=20260927, n_jobs=-1
)
est.fit(y, t, X=X, W=W, groups=groups)
cate = est.effect(X)
lo, hi = est.effect_interval(X)
print(pd.DataFrame({"cate": cate, "lo": lo, "hi": hi}).describe())
```
