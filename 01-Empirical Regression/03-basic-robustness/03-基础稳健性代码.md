## 6. 替代估计量


```stata

```

```stata

```

## 7. 缺失、样本流失与影响点

```stata
misstable summarize $y $x $controls
egen complete_main = rownonmiss($y $x $controls)
tab complete_main

regress $y $x $controls i.year
predict double resid_ols, residuals
predict double leverage_ols, hat
predict double cooksd_ols, cooksd
summarize resid_ols leverage_ols cooksd_ols, detail

gen influential = cooksd_ols > 4/e(N)
regress $y $x $controls i.year if influential == 0, vce(cluster $cluster)
```

## 8. 异方差、序列相关、横截面相关和少量聚类

```stata
regress $y $x $controls i.year
estat hettest
estat imtest, white
estat vif
estat ovtest

xtset panel_id year
xtserial $y $x $controls
xtcsd, pesaran abs

reghdfe $y $x $controls, absorb($fe) vce(cluster $cluster)

```

## 9. Leave-one-cluster-out

```stata
levelsof $cluster, local(cluster_values)
if "$CODEX_PARTIAL_FAILURES"=="" global CODEX_PARTIAL_FAILURES 0
tempname loco_handle
postfile `loco_handle' double omitted_cluster rc beta se N using "03_loco_results.dta", replace
foreach g of local cluster_values {
    preserve
        drop if $cluster == `g'
        capture quietly reghdfe $y $x $controls, absorb($fe) vce(cluster $cluster)
        local run_rc = _rc
        if `run_rc' == 0 {
            local run_beta = _b[$x]
            local run_se   = _se[$x]
            local run_N    = e(N)
        }
        else {
            global CODEX_PARTIAL_FAILURES = $CODEX_PARTIAL_FAILURES + 1
            local run_beta = .
            local run_se   = .
            local run_N    = .
        }
        post `loco_handle' (`g') (`run_rc') (`run_beta') (`run_se') (`run_N')
    restore
}
postclose `loco_handle'
preserve
use "03_loco_results.dta", clear
export delimited using "03_loco_results.csv", replace
restore
display "CODEX_PARTIAL_FAILURES=$CODEX_PARTIAL_FAILURES"
```

## 10. 基准样本一致性

在 02 主回归后先保存样本标记：

```stata
gen byte sample_baseline = e(sample)
count if sample_baseline == 1
```

每个稳健性规格后比较：

```stata
gen byte sample_current = e(sample)
tab sample_baseline sample_current, missing
count if sample_baseline == 1 & sample_current == 0
drop sample_current
```
