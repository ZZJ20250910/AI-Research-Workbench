## A. 常规回归、面板与同步 DID

```stata
global y "被解释变量"
global x "核心解释变量或DID变量"
global controls "控制变量1 控制变量2 控制变量3"
global fe "个体ID 年份"
global cluster "与冲击层级一致的聚类变量"

estimates clear
regress $y $x, vce(cluster $cluster)
estimates store m1
regress $y $x $controls, vce(cluster $cluster)
estimates store m2
reghdfe $y $x, absorb($fe) vce(cluster $cluster)
estimates store m3
reghdfe $y $x $controls, absorb($fe) vce(cluster $cluster)
estimates store m4

esttab m1 m2 m3 m4 using "`temp'/02-baseline.rtf", replace ///
    b(%9.4f) se(%9.4f) star(* 0.10 ** 0.05 *** 0.01) ///
    stats(N r2_a, labels("N" "Adjusted R2"))
```

## H. 因果机器学习基准（Python）

```python
import pandas as pd
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier
from sklearn.model_selection import KFold
from econml.dml import LinearDML

df = pd.read_stata(r"终版数据绝对路径")
y = df["Y"].to_numpy()
t = df["T"].to_numpy()
X = df[["异质性变量1", "异质性变量2"]].to_numpy()
W = df[["控制变量1", "控制变量2"]].to_numpy()
seed = 20260927
cv = KFold(5, shuffle=True, random_state=seed)
est = LinearDML(
    model_y=RandomForestRegressor(n_estimators=500, random_state=seed),
    model_t=RandomForestClassifier(n_estimators=500, random_state=seed),
    discrete_treatment=True, cv=cv, random_state=seed
)
est.fit(y, t, X=X, W=W)
print("ATE", est.ate(X=X))
print("ATE_CI", est.ate_interval(X=X))
```
