## 2. 分组回归与 bootstrap 差异

下面是一项分组维度的完整模板。每增加一个维度，复制整段并单独配置阈值、控制变量和模型；不能默认所有分组共用同一控制变量集合。临时分组变量必须使用短英文名。

```stata
* ---------- 用户映射区：每个分组维度单独配置 ----------
local test_name   "用户确认的分组维度名称"
local split_var   "用户确认的分组依据变量"
local target_var  "x"
local controls    "用户确认的控制变量列表"
local absorb_vars "用户确认的固定效应列表"
local cluster_var "用户确认的聚类变量"
local reps        1000
local seed        20260927

* 若阈值要求按本次基准回归样本计算，先用完全相同的模型锁定e(sample)
quietly reghdfe y `target_var' `controls', ///
    absorb(`absorb_vars') vce(cluster `cluster_var')
capture drop het_sample split_mean gsplit
gen byte het_sample = e(sample)
egen double split_mean = mean(`split_var') if het_sample==1
quietly summarize split_mean, meanonly
replace split_mean = r(mean) if het_sample==1

* 禁止在旧分组变量上replace；源变量缺失时必须保持组别缺失
gen byte gsplit = (`split_var' > split_mean) ///
    if het_sample==1 & !missing(`split_var', split_mean)
assert inlist(gsplit,0,1) if !missing(gsplit)
quietly count if gsplit==0
local n0 = r(N)
quietly count if gsplit==1
local n1 = r(N)

tempname groupregs groupdiff
if "$CODEX_PARTIAL_FAILURES"=="" global CODEX_PARTIAL_FAILURES 0
postfile `groupregs' str80 test byte group double b se p long N int run_status ///
    using "group_regression_results.dta", replace
postfile `groupdiff' str80 test double observed_difference empirical_p ///
    int reps long N0 N1 int run_status using "group_difference_results.dta", replace

* 两组分别估计；每项独立失败，结果始终保留一行
foreach g in 0 1 {
    capture quietly reghdfe y `target_var' `controls' if gsplit==`g', ///
        absorb(`absorb_vars') vce(cluster `cluster_var')
    local rc = _rc
    if `rc'==0 {
        local b  = _b[`target_var']
        local se = _se[`target_var']
        local p  = 2*ttail(e(df_r),abs(`b'/`se'))
        local N  = e(N)
        post `groupregs' ("`test_name'") (`g') (`b') (`se') (`p') (`N') (`rc')
    }
    else {
        global CODEX_PARTIAL_FAILURES = $CODEX_PARTIAL_FAILURES + 1
        post `groupregs' ("`test_name'") (`g') (.) (.) (.) (.) (`rc')
    }
}

* bdiff只接收0/1且无缺失的短英文分组变量；两组都存在才运行
preserve
keep if inlist(gsplit,0,1)
quietly count
local ndiff = r(N)
if `n0'>0 & `n1'>0 {
    capture quietly bdiff, group(gsplit) ///
        model(reghdfe y `target_var' `controls', ///
        absorb(`absorb_vars') cluster(`cluster_var')) ///
        bs reps(`reps') seed(`seed') first detail
    local rc = _rc
    if `rc'==0 {
        * 必须紧跟bdiff保存；后续任何r-class命令都会覆盖它们
        scalar this_diff = r(bdiff)
        scalar this_p = r(p)
        post `groupdiff' ("`test_name'") (this_diff) (this_p) ///
            (`reps') (`n0') (`n1') (`rc')
    }
    else {
        global CODEX_PARTIAL_FAILURES = $CODEX_PARTIAL_FAILURES + 1
        post `groupdiff' ("`test_name'") (.) (.) (`reps') (`n0') (`n1') (`rc')
    }
}
else {
    global CODEX_PARTIAL_FAILURES = $CODEX_PARTIAL_FAILURES + 1
    post `groupdiff' ("`test_name'") (.) (.) (`reps') (`n0') (`n1') (2000)
}
restore

postclose `groupregs'
postclose `groupdiff'

preserve
use "group_regression_results.dta", clear
export delimited using "group_regression_results.csv", replace
restore
preserve
use "group_difference_results.dta", clear
export delimited using "group_difference_results.csv", replace
restore
display "CODEX_PARTIAL_FAILURES=$CODEX_PARTIAL_FAILURES"
```

正式读取只读上述两个小型结果文件。不得从log搜索`Observed difference`或`Empirical p-value`；`bdiff`产生的重复警告留在磁盘，不用于判断运行进度。

## 8. Honest CATE 与验证（Python）

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier
from econml.dml import CausalForestDML

df = pd.read_stata(r"终版数据")
train, test = train_test_split(df, test_size=0.3, random_state=20260927,
                               stratify=df["T"])
features = ["Z1", "Z2", "Z3"]
controls = ["W1", "W2"]

forest = CausalForestDML(
    model_y=RandomForestRegressor(n_estimators=500, min_samples_leaf=10, random_state=20260927),
    model_t=RandomForestClassifier(n_estimators=500, min_samples_leaf=10, random_state=20260927),
    discrete_treatment=True, n_estimators=1000, min_samples_leaf=10,
    cv=5, random_state=20260927, n_jobs=-1
)
forest.fit(train["Y"], train["T"], X=train[features], W=train[controls])
cate = forest.effect(test[features])
lo, hi = forest.effect_interval(test[features])
out = test[["observation_id", "T", "Y"]].copy()
out["cate"] = cate; out["ci_low"] = lo; out["ci_high"] = hi
out["cate_quintile"] = pd.qcut(out["cate"], 5, labels=False, duplicates="drop")
print(out.groupby("cate_quintile")["cate"].agg(["count", "mean", "std"]))
out.to_csv(r"AI-Work\Temp\08-异质性与分布效应\08-cate-validation.csv",
           index=False, encoding="utf-8-sig")
```
